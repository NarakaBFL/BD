CREATE DATABASE IF NOT EXISTS gestion_residuos
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE gestion_residuos;

CREATE TABLE rol (
    id_rol INT UNSIGNED AUTO_INCREMENT,
    nombre_rol VARCHAR(50) NOT NULL,
    descripcion VARCHAR(255),

    CONSTRAINT pk_rol
        PRIMARY KEY (id_rol),

    CONSTRAINT uq_rol_nombre
        UNIQUE (nombre_rol)
) ENGINE = InnoDB;

CREATE TABLE usuario (
    id_usuario INT UNSIGNED AUTO_INCREMENT,
    nombre VARCHAR(60) NOT NULL,
    apellido VARCHAR(60) NOT NULL,
    email VARCHAR(120) NOT NULL,
    contrasena VARCHAR(255) NOT NULL,
    id_rol INT UNSIGNED NOT NULL,
    activo BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT pk_usuario
        PRIMARY KEY (id_usuario),

    CONSTRAINT uq_usuario_email
        UNIQUE (email),

    CONSTRAINT fk_usuario_rol
        FOREIGN KEY (id_rol)
        REFERENCES rol (id_rol)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
) ENGINE = InnoDB;

CREATE TABLE contenedor (
    id_contenedor INT UNSIGNED AUTO_INCREMENT,
    estado ENUM(
        'funcional',
        'roto',
        'desbordado'
    ) NOT NULL DEFAULT 'funcional',

    tipo_residuo VARCHAR(80) NOT NULL,
    capacidad DECIMAL(10,2) NOT NULL,
    ubicacion VARCHAR(150) NOT NULL,
    direccion VARCHAR(180) NOT NULL,

    CONSTRAINT pk_contenedor
        PRIMARY KEY (id_contenedor),

    CONSTRAINT chk_contenedor_capacidad
        CHECK (capacidad > 0)
) ENGINE = InnoDB;

CREATE TABLE camion (
    id_camion INT UNSIGNED AUTO_INCREMENT,
    matricula VARCHAR(20) NOT NULL,

    estado ENUM(
        'disponible',
        'en_servicio',
        'mantenimiento',
        'fuera_de_servicio'
    ) NOT NULL DEFAULT 'disponible',

    capacidad DECIMAL(10,2) NOT NULL,
    id_usuario_asigna INT UNSIGNED NOT NULL,

    CONSTRAINT pk_camion
        PRIMARY KEY (id_camion),

    CONSTRAINT uq_camion_matricula
        UNIQUE (matricula),

    CONSTRAINT chk_camion_capacidad
        CHECK (capacidad > 0),

    CONSTRAINT fk_camion_usuario
        FOREIGN KEY (id_usuario_asigna)
        REFERENCES usuario (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT
) ENGINE = InnoDB;

CREATE TABLE incidencia (
    id_incidencia INT UNSIGNED AUTO_INCREMENT,
    fecha DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    tipo VARCHAR(80) NOT NULL,
    descripcion VARCHAR(500) NOT NULL,

    estado ENUM(
        'abierta',
        'en_curso',
        'resuelta'
    ) NOT NULL DEFAULT 'abierta',

    id_usuario INT UNSIGNED NOT NULL,
    id_contenedor INT UNSIGNED NOT NULL,
    id_camion INT UNSIGNED NULL,

    CONSTRAINT pk_incidencia
        PRIMARY KEY (id_incidencia),

    CONSTRAINT fk_incidencia_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuario (id_usuario)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_incidencia_contenedor
        FOREIGN KEY (id_contenedor)
        REFERENCES contenedor (id_contenedor)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_incidencia_camion
        FOREIGN KEY (id_camion)
        REFERENCES camion (id_camion)
        ON UPDATE CASCADE
        ON DELETE SET NULL
) ENGINE = InnoDB;
